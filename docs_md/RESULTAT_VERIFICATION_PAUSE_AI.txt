COMMANDES SQL UTILES
───────────────────────────────────────────────────────────────────────────

-- Voir tous les trajets EN_COURS avec temps de conduite
SELECT 
    id, point_depart, destination,
    TIMESTAMPDIFF(MINUTE, date_depart, NOW()) / 60.0 AS heures_conduite
FROM trajet 
WHERE statut = 'EN_COURS';

-- Voir les prédictions récentes
SELECT * FROM pause_ai_prediction 
ORDER BY timestamp DESC 
LIMIT 10;

-- Diagnostic d'un trajet spécifique
SELECT 
    t.id, t.point_depart, t.destination,
    TIMESTAMPDIFF(MINUTE, t.date_depart, NOW()) / 60.0 AS heures,
    COUNT(p.id) AS nb_predictions
FROM trajet t
LEFT JOIN pause_ai_prediction p ON p.trajet_id = t.id
WHERE t.id = [VOTRE_ID]
GROUP BY t.id;
